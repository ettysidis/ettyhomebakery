function doGet(e) { const action = e.parameter.action; if (action === "getInventory") { return ContentService .createTextOutput(JSON.stringify(getInventory())) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(JSON.stringify({ success: false })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const data = JSON.parse(e.postData.contents); const action = data.action; if (action === "saveOrder") { return ContentService .createTextOutput(JSON.stringify(saveOrder(data))) .setMimeType(ContentService.MimeType.JSON); } if (action === "updateInventory") { return ContentService .createTextOutput(JSON.stringify(updateInventory(data.inventory))) .setMimeType(ContentService.MimeType.JSON); } return ContentService .createTextOutput(JSON.stringify({ success: false })) .setMimeType(ContentService.MimeType.JSON); } function getInventory() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("מלאי"); const data = sheet.getDataRange().getValues(); let inventory = {}; for (let i = 1; i < data.length; i++) { inventory[data[i][0]] = data[i][1]; } return { success: true, inventory: inventory }; } function saveOrder(params) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const ordersSheet = ss.getSheetByName("הזמנות"); ordersSheet.appendRow([ params.date || new Date().toLocaleString("he-IL"), params.clientName || "", params.clientPhone || "", params.clientAddress || "", params.items || "", params.total || "", params.notes || "", params.clientEmail || "" ]); return { success: true }; } catch (e) { return { success: false, error: e.toString() }; } finally { lock.releaseLock(); } } function updateInventory(newInventory) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("מלאי"); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { const productName = data[i][0]; if (newInventory.hasOwnProperty(productName)) { sheet.getRange(i + 1, 2).setValue(newInventory[productName]); } } return { success: true }; } catch (e) { return { success: false, error: e.toString() }; } finally { lock.releaseLock(); } }
